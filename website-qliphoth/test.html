  • Pattern matching on Result variants for error handling

Example Server Code

/// Handle a single HTTP request
fn handle_request(stream: TcpStream, root_dir: str) {
    let reader = BufReader·new(&mut stream);
    let request_line = BufReader·read_line(reader);

    match request_line {
        Ok(line) => {
            let path = parse_request(line);
            let file_path = if path == "/" {
                f"{root_dir}/index.html"
            } else {
                f"{root_dir}{path}"
            };

            match fs·read_to_string(file_path) {
                Ok(content) => {
                    let mime = get_mime_type(file_path);
                    let response = build_response(200, "OK", mime, content);
                    TcpStream·write_all(stream, response);
                }
                Err(e) => {
                    let response = not_found_response(path);
                    TcpStream·write_all(stream, response);
                }
            }
        }
        Err(e) => print(f"[ERROR] {e}"),
    }
}
